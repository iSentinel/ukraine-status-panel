cmake_minimum_required(VERSION 3.16)
project(usp_artwork_lcd_preflight LANGUAGES C CXX)
add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
add_link_options(-fsanitize=address,undefined)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${USP_PROJECT_ROOT}/tests/lvgl_navigation_host/lv_conf.h" CACHE PATH "" FORCE)
add_subdirectory("${USP_LVGL_ROOT}" lvgl)
add_executable(artwork_lcd main.cpp)
target_include_directories(artwork_lcd PRIVATE "${USP_PROJECT_ROOT}/include")
target_compile_features(artwork_lcd PRIVATE cxx_std_17)
target_link_libraries(artwork_lcd PRIVATE lvgl m)
